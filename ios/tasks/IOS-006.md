## SpectrumView.swift

Draw with `Canvas` — avoid `List` or `Chart` for performance at 23 Hz update rate.

```swift
struct SpectrumView: View {
    let spectrum: [Float]           // fftSize/2 power values in dBFS
    let geqBins: [Int]             // from FFTEngine.geqBins (31 entries)
    let parActive: [Bool]          // from NotchController.parActive (31 entries)
    let threshold: Float           // dB above baseline to show as "alert" color

    var body: some View {
        Canvas { context, size in
            let n = spectrum.count
            let barWidth = size.width / CGFloat(n)
            for i in 0..<n {
                let db = spectrum[i]
                let height = heightForDb(db, in: size.height)
                let rect = CGRect(x: CGFloat(i) * barWidth,
                                  y: size.height - height,
                                  width: barWidth, height: height)
                // Color: red if notched par, yellow if above threshold, green otherwise
                let color = colorForBin(i)
                context.fill(Path(rect), with: .color(color))
            }
        }
    }

    private func heightForDb(_ db: Float, in maxHeight: CGFloat) -> CGFloat {
        // Map –80 dBFS..0 dBFS → 0..maxHeight
        let clamped = max(-80, min(0, db))
        return CGFloat((clamped + 80) / 80) * maxHeight
    }

    private func colorForBin(_ bin: Int) -> Color {
        // Check if this bin belongs to an active notch
        for (j, gbBin) in geqBins.enumerated() {
            if abs(bin - gbBin) <= 1 && parActive[j] { return .red }
        }
        // Check if above a visual threshold (simplified — use raw spectrum level)
        if spectrum[bin] > -20 { return .yellow }
        return .green.opacity(0.7)
    }
}
```

## ContentView.swift

```swift
struct ContentView: View {
    @StateObject private var audio = AudioCaptureEngine(fftEngine: FFTEngine())
    @StateObject private var connection: XR18Connection
    @StateObject private var notchController: NotchController
    @StateObject private var detector: DetectorViewModel

    @AppStorage("xr18_ip")       private var xr18IP = "192.168.0.64"
    @AppStorage("toast_channel") private var channel = 1
    @AppStorage("fx_slot")       private var fxSlot = 4
    @AppStorage("threshold_db")  private var thresholdDb: Double = 20
    @AppStorage("cut_db")        private var cutDb: Double = -6
    @AppStorage("release_sec")   private var releaseSec: Double = 10

    @State private var showConfig = false

    var body: some View {
        VStack(spacing: 0) {
            headerBar
            SpectrumView(spectrum: audio.spectrum,
                         geqBins: detector.geqBins,
                         parActive: notchController.parActive,
                         threshold: Float(thresholdDb))
            frequencyRuler
            notchMarkerRow
            statusBar
        }
        .sheet(isPresented: $showConfig) { ConfigView() }
        .task { await startAll() }
        .onDisappear { stopAll() }
    }
}
```

### DetectorViewModel

Bridges `AudioCaptureEngine.spectrum` → `FeedbackDetector` → `NotchController.update(detectedPar:)`.

```swift
final class DetectorViewModel: ObservableObject {
    private let detector: FeedbackDetector
    private let notchController: NotchController
    let geqBins: [Int]

    init(detector: FeedbackDetector, notchController: NotchController, geqBins: [Int]) { ... }

    func process(spectrum: [Float], baseline: [Float]) {
        let notchedPars = Set(notchController.parActive.enumerated()
            .compactMap { $0.element ? $0.offset + 1 : nil })
        let detectedPar = detector.process(bins: spectrum,
                                            baseline: baseline,
                                            notchedPars: notchedPars)
        notchController.update(detectedPar: detectedPar)
    }
}
```

`BaselineTracker` state (a `[Float]` array initialised to `Array(repeating: -60, count: spectrum.count)`) lives in `DetectorViewModel`; it is updated each frame before calling `detector.process`.

## ConfigView.swift

A `Form` presented as a sheet:

```swift
struct ConfigView: View {
    @AppStorage("xr18_ip")       private var xr18IP = "192.168.0.64"
    @AppStorage("fx_slot")       private var fxSlot = 4
    @AppStorage("threshold_db")  private var thresholdDb: Double = 20
    @AppStorage("cut_db")        private var cutDb: Double = -6
    @AppStorage("release_sec")   private var releaseSec: Double = 10

    var body: some View {
        NavigationStack {
            Form {
                Section("XR18") {
                    TextField("IP Address", text: $xr18IP)
                        .keyboardType(.numbersAndPunctuation)
                    Stepper("FX Slot: \(fxSlot)", value: $fxSlot, in: 1...4)
                }
                Section("Detection") {
                    Slider(value: $thresholdDb, in: 5...40, step: 1) {
                        Text("Threshold: \(Int(thresholdDb)) dB")
                    }
                    Slider(value: $cutDb, in: -15 ... -1, step: 1) {
                        Text("Cut: \(Int(cutDb)) dB")
                    }
                    Slider(value: $releaseSec, in: 2...30, step: 1) {
                        Text("Release: \(Int(releaseSec)) s")
                    }
                }
            }
            .navigationTitle("Settings")
        }
    }
}
```

## Acceptance Criteria

- [ ] App launches and requests microphone permission on first run.
- [ ] `SpectrumView` updates visually at ~23 Hz (no perceptible stuttering at iPhone 16 / iPhone 15 speed).
- [ ] When a 1 kHz tone is played near the mic, the bar at the 1 kHz bin turns yellow.
- [ ] When a narrow 1 kHz tone persists for 3 frames above threshold, the bar turns red and the notch marker (▼) appears.
- [ ] After `releaseSec` quiet seconds, the bar ramps back to green and the ▼ disappears.
- [ ] Config sheet opens, values persist across app restarts.
- [ ] `UIIdleTimerDisabled = YES` prevents screen lock during active session.
- [ ] `connection.disconnect()` and `notchController.restoreAll()` called when app moves to background or view disappears.

## Notes

- The spectrum has `fftSize/2 = 1024` bins. Drawing 1024 bars at 23 Hz via Canvas is fast on iPhone. Do not use `ForEach` or individual `Rectangle` views — Canvas is required.
- `@AppStorage` values are read once at startup by `ContentView`. When the user changes config and dismisses the sheet, call `connection.disconnect()` + `connection.connect()` to reconnect with the new IP.
- The frequency ruler below the spectrum needs only three fixed labels: "20Hz" (left edge), "1kHz" (calculated x position from `geqBins[17]`), "20kHz" (right edge).
- Tap anywhere on the spectrum to call `notchController.restoreAll()` — manual reset for the sound engineer.
