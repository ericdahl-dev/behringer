## Goal

Implement `XR18Connection.swift`: manages the UDP connection to the XR18, sends `/xinfo` handshake and `/xremote` keepalive, and exposes a method to apply/release TEQ notch cuts. Also implement `NotchController.swift`: tracks the 31-band notch state, drives the smooth release ramp, and calls `XR18Connection` to send OSC.

## Files

- `ios/ToastSaver/XR18Connection.swift`
- `ios/ToastSaver/NotchController.swift`

## XR18Connection.swift

```swift
import Network
import Combine
import ToastSaverCore

final class XR18Connection: ObservableObject {
    @Published private(set) var state: ConnectionState = .disconnected

    enum ConnectionState { case disconnected, connecting, connected, failed(String) }

    private let host: String
    private let port: UInt16
    private var connection: NWConnection?
    private var keepaliveTimer: Timer?

    init(host: String, port: UInt16 = 10024)

    /// Send /xinfo and wait for reply. Sets state = .connected on success.
    func connect() async

    /// Stop connection and cancel keepalive timer.
    func disconnect()

    /// Send /fx/{slot}/par/{par:02d} ,f {value} — fire and forget.
    func sendTEQBand(slot: Int, par: Int, value: Float)

    /// Send arbitrary pre-built OSC Data packet.
    func send(_ data: Data)
}
```

### NWConnection setup

```swift
private func makeConnection() {
    let endpoint = NWEndpoint.hostPort(
        host: NWEndpoint.Host(host),
        port: NWEndpoint.Port(rawValue: port)!
    )
    connection = NWConnection(to: endpoint, using: .udp)
    connection?.stateUpdateHandler = { [weak self] newState in
        DispatchQueue.main.async {
            switch newState {
            case .ready:    self?.state = .connected; self?.startKeepalive()
            case .failed(let err): self?.state = .failed(err.localizedDescription)
            default: break
            }
        }
    }
    connection?.start(queue: .global(qos: .userInteractive))
}
```

### /xinfo handshake

```swift
func connect() async {
    await MainActor.run { state = .connecting }
    makeConnection()
    // Wait up to 2 seconds for .ready
    try? await Task.sleep(for: .seconds(2))
    guard case .connected = state else {
        await MainActor.run { state = .failed("No reply from \(host)") }
        return
    }
    send(OSCPacket.xinfo())
}
```

### Keepalive

```swift
private func startKeepalive() {
    keepaliveTimer = Timer.scheduledTimer(withTimeInterval: 9, repeats: true) { [weak self] _ in
        self?.send(OSCPacket.xremote())
    }
    send(OSCPacket.xremote())
}
```

### sendTEQBand

```swift
func sendTEQBand(slot: Int, par: Int, value: Float) {
    send(OSCPacket.teqBand(slot: slot, par: par, value: value))
}

func send(_ data: Data) {
    connection?.send(content: data, completion: .idempotent)
}
```

## NotchController.swift

```swift
import Combine
import ToastSaverCore

struct BandState {
    var active: Bool = false
    var cutValue: Float = 0.5          // OSC float sent when notched
    var notchedAt: Date = .distantPast
    var releaseHold: Int = 0           // consecutive quiet frames
}

final class NotchController: ObservableObject {
    /// par_active[i] = true if GEQ par i+1 is notched. Updated on main thread.
    @Published private(set) var parActive: [Bool] = Array(repeating: false, count: 31)

    private var bands: [BandState] = Array(repeating: BandState(), count: 31)
    private let connection: XR18Connection
    private let fxSlot: Int
    private let cutDb: Float
    private let releaseSeconds: Float
    private let updateRate: Float = 20.0   // approximate FFT frames/second

    init(connection: XR18Connection, fxSlot: Int = 4, cutDb: Float = -6, releaseSeconds: Float = 10)

    /// Call once per FFT frame with the detected GEQ par (or nil if no detection).
    func update(detectedPar: Int?)

    /// Restore all active notches to flat. Call on disconnect or app resign.
    func restoreAll()
}
```

### update() algorithm

Port the two-pass logic from `run_detection()` in `src/XAir_ToastSaver.c`:

```swift
func update(detectedPar: Int?) {
    let releaseFrames = Int(releaseSeconds * updateRate)
    let cutValue = OSCPacket.geqFloat(cutDb)

    // Pass 1: release ramp for all active bands
    for j in 0..<31 {
        guard bands[j].active else { continue }
        let par = j + 1
        let isCurrentTarget = detectedPar == par

        if !isCurrentTarget {
            bands[j].releaseHold += 1
            let t = min(Float(bands[j].releaseHold) / Float(releaseFrames), 1.0)
            let rampValue = bands[j].cutValue + (0.5 - bands[j].cutValue) * t
            connection.sendTEQBand(slot: fxSlot, par: par, value: rampValue)

            if bands[j].releaseHold >= releaseFrames {
                bands[j].active = false
                bands[j].releaseHold = 0
            }
        } else {
            if bands[j].releaseHold > 0 {
                // Re-triggered mid-fade — snap back
                bands[j].releaseHold = 0
                connection.sendTEQBand(slot: fxSlot, par: par, value: bands[j].cutValue)
            }
        }
    }

    // Pass 2: apply new notch
    if let par = detectedPar {
        let j = par - 1
        if !bands[j].active {
            connection.sendTEQBand(slot: fxSlot, par: par, value: cutValue)
            bands[j] = BandState(active: true, cutValue: cutValue, notchedAt: Date(), releaseHold: 0)
        }
    }

    // Publish state to SwiftUI
    let active = bands.map(\.active)
    DispatchQueue.main.async { self.parActive = active }
}

func restoreAll() {
    for j in 0..<31 where bands[j].active {
        connection.sendTEQBand(slot: fxSlot, par: j + 1, value: 0.5)
        bands[j].active = false
    }
    DispatchQueue.main.async { self.parActive = Array(repeating: false, count: 31) }
}
```

## Acceptance Criteria

- [ ] `XR18Connection.connect()` to a reachable XR18 sets state `.connected` within 2 seconds.
- [ ] `XR18Connection.connect()` to an unreachable IP times out and sets state `.failed(...)`.
- [ ] Keepalive timer sends `/xremote` every 9 seconds (verify with a packet capture or Serial log on the mixer side).
- [ ] `sendTEQBand(slot:4, par:18, value:0.3)` produces exactly the same byte sequence as the CLI tool for `/fx/4/par/18 ,f 0.3`.
- [ ] `NotchController.update(detectedPar:18)` triggers for `confirmFrames` consecutive frames → sends a notch OSC packet on the last frame.
- [ ] `NotchController.update(detectedPar:nil)` for `releaseSeconds * 20` frames → TEQ band smoothly ramps back to 0.5 and `parActive[17]` becomes false.
- [ ] Re-triggering a fading band snaps back to `cutValue` immediately.
- [ ] `restoreAll()` sends `0.5` for every active band and clears `parActive`.
- [ ] `disconnect()` cancels the NWConnection and keepalive timer without crashing.

## Notes

- `NWConnection` on UDP is connectionless — it will go `.ready` immediately even if the XR18 is unreachable. The real handshake is the `/xinfo` reply, but since we're not subscribing to RTA (we have our own FFT), we send `/xinfo` as a liveness check only. If you want to verify the XR18 is present, listen for any incoming packet within 1 second.
- `NotchController.update()` is called on the FFT dispatch queue — never on the main thread. All UI-facing state must be dispatched to main.
- `fxSlot` default is 4 to match the CLI tool default. Expose it in the config screen (IOS-006).

## Handoff (2026-06-27)

The OSC encoder already exists in `ToastSaverCore` (shipped under IOS-001):
`OSCMessage(address, .float/.int32/.string)` + `OSCMessage.geqBand(slot:par:dB:)` —
Foundation-free, big-endian, 4-byte-aligned, tested. **Reuse it for the notch
packets; don't write a second encoder.** Not built yet: `xinfo()` / `xremote()`
handshake packets (add in IOS-002 or here). If IOS-002 reconciles the OSC API to
`OSCPacket.teqBand(…)->Data`, use that name instead — confirm which API won before
wiring `NotchController`.
